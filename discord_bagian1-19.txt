# ---------- Cache sederhana ----------
_cache = {}
CACHE_TTL = 60


def surf_run(cmd: str):
    now = time.time()
    if cmd in _cache and now - _cache[cmd]["time"] < CACHE_TTL:
        return _cache[cmd]["data"]
    try:
        r = subprocess.run(cmd, shell=True, capture_output=True, text=True, timeout=30)
        data = json.loads(r.stdout) if r.stdout else {"error": r.stderr[:300]}
    except Exception as e:
        data = {"error": str(e)}
    _cache[cmd] = {"data": data, "time": now}
    return data


def _fmt_ts(unix_ts):
    try:
        if not unix_ts:
            return None
        return time.strftime("%Y-%m-%d", time.gmtime(int(unix_ts)))
    except Exception:
        return None


def _q(text: str) -> str:
    """Escape string untuk dipakai aman di shell command."""
    return shlex.quote(text)


# ---------- Extractors (rapikan hasil mentah Surf) ----------

def extract_fundraising(raw):
    items = raw.get("data", []) if isinstance(raw, dict) else []
    result = []
    for item in items[:6]:
        title = item.get("title")
        if not title:
            continue
        result.append({
            "title": title,
            "summary": item.get("summary"),
            "importance": item.get("importance"),
            "event_date": _fmt_ts(item.get("event_at")),
            "source": (item.get("source") or {}).get("name"),
            "source_url": (item.get("source") or {}).get("url"),
        })
    return result


def extract_airdrop(raw):
    items = raw.get("data", []) if isinstance(raw, dict) else []
    result = []
    for item in items[:20]:
        name = item.get("project_name")
        if not name:
            continue
        task_summary = item.get("task_summary") or {}
        tasks = item.get("tasks") or []
        # Ambil link partisipasi dari task pertama yang statusnya OPEN
        link = None
        for t in tasks:
            if t.get("status") == "OPEN" and t.get("external_link"):
                link = t.get("external_link")
                break
        entry = {
            "project_name": name,
            "coin_symbol": item.get("coin_symbol"),
            "status": item.get("status"),
            "reward_type": item.get("reward_type"),
            "reward_date": _fmt_ts(item.get("reward_date")),
            "total_raise_usd": item.get("total_raise") or None,
            "xscore": item.get("xscore") or None,
            "open_tasks": task_summary.get("open"),
            "total_tasks": task_summary.get("total"),
            "logo_url": item.get("logo_url") or None,
        }
        if link:
            entry["link"] = link
        result.append(entry)
    return result


def extract_social_people(raw, limit=3):
    """Sesuai skema asli surf search-social-people. TIDAK ada field verifikasi
    resmi — hasil ini cuma 'kemungkinan akun terkait', bukan konfirmasi tim."""
    items = raw.get("data", []) if isinstance(raw, dict) else []
    result = []
    for item in items[:limit]:
        if not isinstance(item, dict):
            continue
        handle = item.get("handle")
        if not handle:
            continue
        result.append({
            "handle": handle,
            "name": item.get("name"),
            "bio": item.get("bio"),
            "followers_count": item.get("followers_count"),
        })
    return result


def extract_social_posts(raw, limit=6):
    items = raw.get("data", []) if isinstance(raw, dict) else []
    result = []
    for item in items[:limit]:
        if not isinstance(item, dict):
            continue
        text = item.get("text")
        if not text:
            continue
        author = item.get("author") or {}
        stats = item.get("stats") or {}
        result.append({
            "author_handle": author.get("handle"),
            "text": text[:280],
            "posted_at": _fmt_ts(item.get("created_at")),
            "likes": stats.get("likes"),
            "views": stats.get("views"),
            "url": item.get("url"),
        })
    return result


def extract_signal_tokens(raw, limit=6):
    """Sesuai skema asli surf signal-token-of-the-day / signal-token-of-week."""
    items = raw.get("data", []) if isinstance(raw, dict) else []
    result = []
    for item in items[:limit]:
        if not isinstance(item, dict):
            continue
        name = item.get("name")
        if not name:
            continue
        price_chg = item.get("price_change_percent") or {}
        core_state = item.get("core_state") or {}
        ai_summary = item.get("ai_summary") or {}
        # ai_summary formatnya {<any>: string} — ambil value pertama yang ada
        summary_text = next(iter(ai_summary.values()), None) if ai_summary else None

        entry = {
            "name": name,
            "symbol": item.get("symbol"),
            "rank": item.get("rank"),
            "price_usd": item.get("price"),
            "change_24h_percent": round(price_chg.get("24h"), 1) if isinstance(price_chg.get("24h"), (int, float)) else None,
            "change_7d_percent": round(price_chg.get("7d"), 1) if isinstance(price_chg.get("7d"), (int, float)) else None,
            "heat_score": round(item.get("heat_score"), 1) if isinstance(item.get("heat_score"), (int, float)) else None,
            "state": core_state.get("state"),
            "state_reason": core_state.get("reason"),
            "ai_summary": summary_text,
        }
        # Buang key yang value-nya None biar tidak berisik
        entry = {k: v for k, v in entry.items() if v is not None}
        result.append(entry)
    return result


def extract_project_search(raw, limit=6):
    """Sesuai skema asli surf search-project — TIDAK ada field website resmi."""
    items = raw.get("data", []) if isinstance(raw, dict) else []
    result = []
    for item in items[:limit]:
        if not isinstance(item, dict):
            continue
        name = item.get("name")
        if not name:
            continue
        entry = {
            "name": name,
            "symbol": item.get("symbol"),
            "description": item.get("description"),
            "chains": item.get("chains") or None,
            "tags": item.get("tags") or None,
        }
        entry = {k: v for k, v in entry.items() if v}
        result.append(entry)
    return result


def simplify(data, limit=6):
    if isinstance(data, list):
        data = data[:limit]
        simplified = []
        for item in data:
            if not isinstance(item, dict):
                continue
            project = item.get("project", {}) if isinstance(item.get("project"), dict) else {}
            name = item.get("name") or project.get("name")
            if not name:
                continue
            raw_value = item.get("value") or item.get("score") or item.get("mindshare")
            entry = {
                "name": name,
                "symbol": item.get("symbol") or project.get("symbol"),
                "score": round(raw_value, 1) if isinstance(raw_value, (int, float)) else raw_value,
            }
            for key in ["category", "chain", "market_cap", "launch_date", "status"]:
                val = item.get(key) or project.get(key)
                if val:
                    entry[key] = val
            simplified.append(entry)
        return simplified
    return data
