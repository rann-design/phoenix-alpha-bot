

def execute_tool(name: str, args: dict):
    """Jalankan command Surf sesuai tool yang dipanggil AI."""
    query = (args or {}).get("query", "").strip()

    if name == "search_fundraising":
        cmd = "surf search-fundraising --limit 10 -o json"
        if query:
            cmd += f" --q {_q(query)}"
        return extract_fundraising(surf_run(cmd))

    if name == "search_airdrop":
        cmd = "surf search-airdrop --limit 20 --phase active,claimable --include-tasks -o json"
        if query:
            cmd += f" --q {_q(query)}"
        return extract_airdrop(surf_run(cmd))

    if name == "search_social_posts":
        if not query:
            return {"error": "query wajib diisi"}
        min_faves = (args or {}).get("min_faves", 20)
        cmd = f"surf search-social-posts --q {_q(query)} --min_faves {int(min_faves)} --limit 8 -o json"
        return extract_social_posts(surf_run(cmd))

    if name == "search_social_people":
        if not query:
            return {"error": "query wajib diisi"}
        cmd = f"surf search-social-people --q {_q(query)} --limit 5 -o json"
        return extract_social_people(surf_run(cmd))

    if name == "get_trending_projects":
        return simplify(surf_run("surf signal-projects -o json -f body.data"))

    if name == "get_social_ranking":
        return simplify(surf_run("surf social-ranking -o json -f body.data"))

    if name == "search_news":
        if not query:
            return {"error": "query wajib diisi"}
        cmd = f"surf search-news --query {_q(query)} -o json -f body.data"
        return simplify(surf_run(cmd))

    if name == "search_project":
        if not query:
            return {"error": "query wajib diisi"}
        cmd = f"surf search-project --q {_q(query)} -o json"
        return extract_project_search(surf_run(cmd))

    if name == "get_newest_airdrops":
        # phase=active saja (POTENTIAL+CONFIRMED) = baru diumumkan, belum snapshot
        cmd = "surf search-airdrop --limit 20 --phase active --sort-by last_status_update --order desc --include-tasks -o json"
        return extract_airdrop(surf_run(cmd))

    if name == "get_token_of_the_day":
        return extract_signal_tokens(surf_run("surf signal-token-of-the-day -o json"))

    if name == "get_token_of_the_week":
        return extract_signal_tokens(surf_run("surf signal-token-of-week -o json"))

    if name == "search_fund":
        if not query:
            return {"error": "query wajib diisi"}
        cmd = f"surf search-fund --q {_q(query)} -o json -f body.data"
        return simplify(surf_run(cmd))

    return {"error": f"tool '{name}' tidak dikenal"}


# Kata kunci yang mengindikasikan pertanyaan BUTUH pencarian data
_RESEARCH_KEYWORDS = [
    "proyek", "project", "airdrop", "early", "baru", "gem", "potential",
    "potensial", "fundraising", "pendanaan", "trending", "tren", "sinyal",
    "signal", "token", "cari", "gali", "cek", "rekomendasi",
]


def _needs_research(msg: str) -> bool:
    msg = msg.lower()
    return any(kw in msg for kw in _RESEARCH_KEYWORDS)


# ---------- Agent loop ----------

async def run_agent(user_msg: str) -> str:
    messages = [
        {"role": "system", "content": SYSTEM_PROMPT},
        {"role": "user", "content": user_msg},
    ]
    force_tool_first_round = _needs_research(user_msg)

    for round_num in range(MAX_TOOL_ROUNDS):
        choice = "required" if (round_num == 0 and force_tool_first_round) else "auto"

        try:
            response = await asyncio.to_thread(
                client.chat.completions.create,
                model=MODEL,
                messages=messages,
                tools=TOOLS,
                tool_choice=choice,
                max_tokens=1200,
                temperature=0.4,
            )
        except Exception as e:
            err_text = str(e)
            print("Groq API error:", err_text)

            if "tool_use_failed" in err_text or "Failed to call a function" in err_text:
                print("Fallback: retry tanpa tool calling...")
                try:
                    fallback = await asyncio.to_thread(
                        client.chat.completions.create,
                        model=MODEL,
                        messages=[
                            {"role": "system", "content": SYSTEM_PROMPT + (
                                "\n\nCATATAN: pencarian data real-time sedang bermasalah. "
                                "Jawab pertanyaan user dengan pengetahuan umummu tentang "
                                "Web3/crypto, dan jelaskan dengan jujur bahwa kamu tidak "
                                "bisa akses data terbaru saat ini — sarankan user coba lagi "
                                "beberapa saat lagi atau pertanyaan lebih spesifik."
                            )},
                            {"role": "user", "content": user_msg},
                        ],
                        max_tokens=1000,
                        temperature=0.4,
                    )
                    return fallback.choices[0].message.content or "Maaf, pencarian sedang bermasalah."
                except Exception as e2:
                    return f"Maaf, pencarian sedang bermasalah dan fallback juga gagal: {e2}"

            return f"Maaf, terjadi error saat menghubungi AI: {err_text[:300]}"

        msg = response.choices[0].message

        if not msg.tool_calls:
            return msg.content or "Maaf, tidak ada jawaban."

        messages.append({
            "role": "assistant",
            "content": msg.content or "",
            "tool_calls": [
                {
                    "id": tc.id,
                    "type": "function",
                    "function": {"name": tc.function.name, "arguments": tc.function.arguments},
                }
                for tc in msg.tool_calls
            ],
        })

        for tc in msg.tool_calls:
            try:
                args = json.loads(tc.function.arguments) if tc.function.arguments else {}
            except Exception:
                args = {}
            print(f"Tool dipanggil: {tc.function.name}({args})")
            try:
                result = await asyncio.to_thread(execute_tool, tc.function.name, args)
            except Exception as e:
                result = {"error": f"gagal jalankan tool: {e}"}
            messages.append({
                "role": "tool",
                "tool_call_id": tc.id,
                "content": json.dumps(result)[:3000],
            })

    return "Maaf, pencarian terlalu kompleks dan belum selesai. Coba pertanyaan yang lebih spesifik."

class AlphaView(discord.ui.View):
    """Card interaktif untuk browsing airdrop terbaru, satu per satu."""

    # Warna & label badge sesuai lifecycle status airdrop (data asli dari Surf)
    STATUS_STYLE = {
        "POTENTIAL": (discord.Color.blue(), "🆕 BARU TERDETEKSI"),
        "CONFIRMED": (discord.Color.green(), "✅ DIKONFIRMASI"),
        "SNAPSHOT": (discord.Color.gold(), "📸 SNAPSHOT"),
        "VERIFICATION": (discord.Color.orange(), "🔐 VERIFIKASI"),
        "REWARD_AVAILABLE": (discord.Color.purple(), "🎁 SIAP DIKLAIM"),
        "DISTRIBUTED": (discord.Color.greyple(), "📦 SELESAI"),
    }

    def __init__(self, items):
        super().__init__(timeout=300)  # tombol nonaktif otomatis setelah 5 menit
        self.items = items
        self.index = 0

    def build_embed(self) -> discord.Embed:
        item = self.items[self.index]
        status = item.get("status", "")
        color, badge = self.STATUS_STYLE.get(status, (discord.Color.default(), None))

        title = item.get("project_name", "Unknown")
        symbol = item.get("coin_symbol")
        if symbol:
            title += f" ({symbol})"

        embed = discord.Embed(
            title=title,
            description=f"**{badge}**" if badge else None,
            color=color,
        )

        if item.get("logo_url"):
            embed.set_thumbnail(url=item["logo_url"])

        # Baris 1: Reward & tanggal
        if item.get("reward_type"):
            embed.add_field(name="🎁 Reward", value=item["reward_type"], inline=True)
        if item.get("reward_date"):
            embed.add_field(name="📅 Tanggal", value=item["reward_date"], inline=True)
        embed.add_field(name="\u200b", value="\u200b", inline=True)  # spacer biar rapi 3 kolom

        # Baris 2: Dana & skor sosial
        if item.get("total_raise_usd"):
            embed.add_field(name="💰 Dana Terkumpul", value=f"${item['total_raise_usd']:,}", inline=True)
        if item.get("xscore"):
            embed.add_field(name="📊 Skor Sosial", value=str(item["xscore"]), inline=True)
        if item.get("open_tasks") is not None:
            embed.add_field(
                name="✅ Task Terbuka",
                value=f"{item.get('open_tasks')}/{item.get('total_tasks', '?')}",
                inline=True,
            )

        # Sumber/link resmi — SELALU ditampilkan eksplisit, jujur kalau tidak ada
        # (bukan disembunyikan), supaya user tahu apa yang bisa diverifikasi sendiri
        if item.get("link"):
            embed.add_field(
                name="🔗 Sumber / Link Resmi (verifikasi sendiri di sini)",
                value=item["link"],
                inline=False,
            )
        else:
            embed.add_field(
                name="🔗 Sumber / Link Resmi",
                value="Tidak tersedia di data Surf — cari manual sebelum ikut serta.",
                inline=False,
            )

        embed.set_footer(
            text=f"Proyek {self.index + 1}/{len(self.items)} • Data agregasi Surf API "
            f"(sumber publik) • Bukan saran investasi — DYOR"
        )
        return embed

    @discord.ui.button(label="🔄 Lain", style=discord.ButtonStyle.secondary)
    async def next_item(self, interaction: discord.Interaction, button: discord.ui.Button):
        self.index = (self.index + 1) % len(self.items)
        await interaction.response.edit_message(embed=self.build_embed(), view=self)

    @discord.ui.button(label="ℹ️ Detail Lengkap", style=discord.ButtonStyle.primary)
    async def detail_lengkap(self, interaction: discord.Interaction, button: discord.ui.Button):
        await interaction.response.defer()  # kasih waktu, panggilan ini butuh 2 request tambahan

        item = self.items[self.index]
        name = item.get("project_name", "")

        # Lazy-load: cuma cari sekarang, buat item yang lagi ditampilkan aja.
        # Pakai asyncio.to_thread supaya subprocess Surf CLI tidak nge-block
        # event loop bot (yang bikin koneksi Discord putus & interaksi gagal)
        try:
            project_info = await asyncio.to_thread(execute_tool, "search_project", {"query": name})
        except Exception:
            project_info = []
        try:
            people_info = await asyncio.to_thread(execute_tool, "search_social_people", {"query": name})
        except Exception:
            people_info = []

        embed = self.build_embed()  # mulai dari card dasar yang sudah ada

        if isinstance(project_info, list) and project_info:
            desc = project_info[0].get("description")
            if desc:
                embed.description = (embed.description + "\n\n" if embed.description else "") + desc
            chains = project_info[0].get("chains")
            if chains:
                embed.add_field(name="⛓️ Chain", value=", ".join(chains), inline=False)

        if isinstance(people_info, list) and people_info:
            lines = []
            for p in people_info[:3]:
                followers = p.get("followers_count")
                followers_str = f" ({followers:,} followers)" if followers else ""
                lines.append(f"@{p['handle']}{followers_str}")
            embed.add_field(
                name="👤 Kemungkinan Akun Terkait (belum terverifikasi resmi)",
                value="\n".join(lines),
                inline=False,
            )

        if not (project_info or people_info):
            embed.add_field(name="ℹ️ Detail Tambahan", value="Tidak ditemukan info tambahan.", inline=False)

        await interaction.edit_original_response(embed=embed, view=self)

# ---------- Shortcut emoji (kirim emoji ini SAJA sebagai pesan) ----------
# Kirim salah satu emoji ini tanpa teks lain → bot otomatis jalankan tool terkait
EMOJI_SHORTCUTS = {
    "🔍": "get_newest_airdrops",       # airdrop baru
    "🛰️": "get_newest_airdrops",       # alias lain buat airdrop baru
    "📈": "get_trending_projects",      # proyek trending
    "💰": "search_fundraising",         # berita fundraising terbaru
}


async def handle_emoji_shortcut(message, emoji: str):
    """Jalankan tool yang sesuai emoji, tampilkan sebagai Embed kalau memungkinkan."""
    tool_name = EMOJI_SHORTCUTS.get(emoji)
    if not tool_name:
        return False

    await message.channel.send(f"{emoji} Mencari data...")
    try:
        result = await asyncio.to_thread(execute_tool, tool_name, {})
    except Exception as e:
        await message.channel.send(f"Gagal ambil data: {e}")
        return True

    if not result or (isinstance(result, dict) and result.get("error")):
        await message.channel.send("Tidak ada data ditemukan saat ini.")
        return True

    # Kalau hasilnya list airdrop, pakai card interaktif yang sudah ada
    if tool_name == "get_newest_airdrops" and isinstance(result, list):
        view = AlphaView(result)
        await message.channel.send(embed=view.build_embed(), view=view)
    else:
        # Untuk tool lain, tampilkan ringkas sebagai teks biasa
        lines = []
        for item in result[:5] if isinstance(result, list) else []:
            name = item.get("name") or item.get("title") or "Unknown"
            extra = item.get("score") or item.get("summary") or ""
            lines.append(f"• **{name}** — {extra}" if extra else f"• **{name}**")
        text = "\n".join(lines) if lines else "Tidak ada data relevan."
        await message.channel.send(text)

    return True


@bot.event
async def on_message(message):
    if message.author.bot:
        return

    # Cek dulu apakah pesan cuma berisi emoji shortcut (tanpa teks lain)
    stripped = message.content.strip()
    if stripped in EMOJI_SHORTCUTS:
        handled = await handle_emoji_shortcut(message, stripped)
        if handled:
            return

    # Biarkan command !start / !stop / !alpha diproses dulu
    await bot.process_commands(message)
    if message.content.startswith(bot.command_prefix):
        return  # sudah ditangani sebagai command, jangan diproses AI lagi

    print("Pesan diterima:", message.content)
    try:
        answer = await run_agent(message.content)
        await send_long_message(message.channel, answer)
    except Exception as e:
        print("ERROR:", e)
        await message.channel.send(f"Terjadi error: {e}")

def load_subscribers() -> set:
    try:
        with open(SUBSCRIBERS_FILE, "r") as f:
            return set(json.load(f))
    except Exception:
        return set()


def save_subscribers(subs: set):
    with open(SUBSCRIBERS_FILE, "w") as f:
        json.dump(list(subs), f)


@tasks.loop(minutes=AUTO_UPDATE_INTERVAL_MINUTES)
async def auto_update_loop():
    subs = load_subscribers()
    if not subs:
        return
    print(f"Menjalankan auto-update untuk {len(subs)} channel...")

    # Gantian kategori tiap kali jalan — dipilih acak dari 2 kategori yang didukung
    category_key = random.choice(list(CATEGORY_INFO.keys()))
    category = CATEGORY_INFO[category_key]

    try:
        raw = await asyncio.to_thread(surf_run, category["cmd"])
        items = extract_airdrop(raw)
    except Exception as e:
        print("Auto-update gagal ambil data:", e)
        return

    if not items:
        print(f"Tidak ada data untuk kategori {category_key}, skip update kali ini.")
        return

    # Ambil 1 proyek acak dari hasil, biar variatif tiap kali jalan
    chosen = random.choice(items[:10])
    view = AlphaView([chosen])  # tetap bisa pakai tombol "Detail Lengkap"

    for channel_id in subs:
        try:
            channel = bot.get_channel(channel_id) or await bot.fetch_channel(channel_id)
            await channel.send(f"**{category['label']} — Update Otomatis**\n{category['explain']}")
            await channel.send(embed=view.build_embed(), view=view)
        except Exception as e:
            print(f"Gagal kirim ke channel {channel_id}: {e}")
